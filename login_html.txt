    <script>
        AOS.init({
            duration: 800,
            once: true,
            offset: 50,
        });
        
        // Dark mode logic
        const themeToggleBtn = document.getElementById('theme-toggle');
        const themeToggleDarkIcon = document.getElementById('theme-toggle-dark-icon');
        const themeToggleLightIcon = document.getElementById('theme-toggle-light-icon');

        // Change the icons inside the button based on previous settings
        if (localStorage.getItem('theme') === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
            themeToggleLightIcon.classList.remove('hidden');
        } else {
            themeToggleDarkIcon.classList.remove('hidden');
        }

        if (themeToggleBtn) {
            themeToggleBtn.addEventListener('click', function() {
                // toggle icons inside button
                themeToggleDarkIcon.classList.toggle('hidden');
                themeToggleLightIcon.classList.toggle('hidden');

                // if set via local storage previously
                if (localStorage.getItem('theme')) {
                    if (localStorage.getItem('theme') === 'light') {
                        document.documentElement.classList.add('dark');
                        localStorage.setItem('theme', 'dark');
                    } else {
                        document.documentElement.classList.remove('dark');
                        localStorage.setItem('theme', 'light');
                    }
                // if NOT set via local storage previously
                } else {
                    if (document.documentElement.classList.contains('dark')) {
                        document.documentElement.classList.remove('dark');
                        localStorage.setItem('theme', 'light');
                    } else {
                        document.documentElement.classList.add('dark');
                        localStorage.setItem('theme', 'dark');
                    }
                }
            });
        }

        // AI Chat Modal Logic
        const aiChatModal = document.getElementById('aiChatModal');
        const openAiChatBtn = document.getElementById('openAiChat');
        const closeAiChatBtn = document.getElementById('closeAiChat');
        const aiChatForm = document.getElementById('aiChatForm');
        const aiChatInput = document.getElementById('aiChatInput');
        const aiChatMessages = document.getElementById('aiChatMessages');
        
        window.openAIChat = function(initialMessage = null) {
            aiChatModal.classList.remove('translate-y-8', 'opacity-0', 'pointer-events-none');
            openAiChatBtn.classList.add('scale-0', 'opacity-0');
            
            if (initialMessage) {
                aiChatInput.value = initialMessage;
                aiChatForm.dispatchEvent(new Event('submit'));
            }
        };

        if (openAiChatBtn && aiChatModal && closeAiChatBtn) {
            openAiChatBtn.addEventListener('click', () => window.openAIChat());
            closeAiChatBtn.addEventListener('click', () => {
                aiChatModal.classList.add('translate-y-8', 'opacity-0', 'pointer-events-none');
                openAiChatBtn.classList.remove('scale-0', 'opacity-0');
            });
        }
        
        window.aiChatHistory = [];
        
        function formatMarkdown(text) {
            return text
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                .replace(/\*(.*?)\*/g, '<em>$1</em>')
                .replace(/\n/g, '<br>');
        }

        if (aiChatForm) {
            aiChatForm.addEventListener('submit', async (e) => {
                e.preventDefault();
                const message = aiChatInput.value.trim();
                if (!message) return;
                
                window.aiChatHistory.push({role: 'user', content: message});
                
                aiChatMessages.innerHTML += `
                    <div class="flex items-start gap-3 flex-row-reverse">
                        <div class="w-8 h-8 rounded-full bg-slate-200 dark:bg-slate-700 flex items-center justify-center text-slate-600 dark:text-white flex-shrink-0 mt-1 shadow-sm transition-colors">
                            <i class="fa-solid fa-user text-xs"></i>
                        </div>
                        <div class="bg-blue-600 text-white p-3 rounded-2xl rounded-tr-sm text-sm shadow-sm max-w-[80%]">
                            ${message}
                        </div>
                    </div>
                `;
                aiChatInput.value = '';
                aiChatMessages.scrollTop = aiChatMessages.scrollHeight;
                
                const responseId = 'resp-' + Date.now();
                aiChatMessages.innerHTML += `
                    <div class="flex items-start gap-3">
                        <div class="w-8 h-8 rounded-full bg-blue-100 dark:bg-blue-500/20 border border-blue-200 dark:border-blue-500/30 flex items-center justify-center text-blue-600 dark:text-blue-400 flex-shrink-0 mt-1">
                            <i class="fa-solid fa-robot text-xs"></i>
                        </div>
                        <div id="${responseId}" class="bg-slate-100 dark:bg-slate-800 text-slate-800 dark:text-slate-200 p-3 rounded-2xl rounded-tl-sm text-sm border border-slate-200 dark:border-slate-700 shadow-sm max-w-[80%] leading-relaxed markdown-body transition-colors">
                            <i class="fa-solid fa-circle-notch fa-spin"></i> Grok is thinking...
                        </div>
                    </div>
                `;
                aiChatMessages.scrollTop = aiChatMessages.scrollHeight;
                
                try {
                    const response = await fetch('/ai/chat/', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({ messages: window.aiChatHistory })
                    });
                    
                    const reader = response.body.getReader();
                    const decoder = new TextDecoder('utf-8');
                    let fullResponse = "";
                    let isFirstChunk = true;
                    const respElement = document.getElementById(responseId);
                    
                    while (true) {
                        const { value, done } = await reader.read();
                        if (done) break;
                        
                        const chunk = decoder.decode(value, {stream: true});
                        const lines = chunk.split('\n');
                        
                        for (let line of lines) {
                            if (line.startsWith('data: ')) {
                                const dataStr = line.substring(6);
                                if (dataStr === '[DONE]') continue;
                                
                                try {
                                    const data = JSON.parse(dataStr);
                                    if (data.content) {
                                        if (isFirstChunk) {
                                            respElement.innerHTML = '';
                                            isFirstChunk = false;
                                        }
                                        fullResponse += data.content;
                                        respElement.innerHTML = formatMarkdown(fullResponse);
                                        aiChatMessages.scrollTop = aiChatMessages.scrollHeight;
                                    }
                                } catch (e) {}
                            }
                        }
                    }
                    
                    window.aiChatHistory.push({role: 'assistant', content: fullResponse});
                    
                } catch (err) {
                    const respElement = document.getElementById(responseId);
                    respElement.classList.add('text-red-600', 'dark:text-red-400');
                    respElement.innerHTML = '<i class="fa-solid fa-triangle-exclamation text-xs"></i> Connection error. Please try again.';
                }
            });
        }
